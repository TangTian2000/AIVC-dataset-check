adata = sc.AnnData(X.T, obs, var)

adata.var.index.name = 'gene_symbol'

adata.obs['perturbation'] = [x.replace('DMSO_', '').replace('_DMSO', '').replace('DMSO', 'control') for x in adata.obs.perturbation]
adata.obs.index.name = 'cell_barcode'
adata.obs.rename({

}, axis=1, inplace=True)
